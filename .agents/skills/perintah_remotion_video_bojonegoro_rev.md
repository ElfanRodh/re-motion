## 4. Contoh Kode Komponen Gaya Neo-Brutalist Siap Pakai (`.tsx`)

Anda dapat langsung menyalin komponen-komponen inti bergaya video referensi ini ke dalam proyek Remotion Anda:

```tsx
import React from 'react';
import {
  AbsoluteFill,
  interpolate,
  spring,
  useCurrentFrame,
  useVideoConfig,
} from 'remotion';

const INK = '#1E1B4B'; // Outline gelap khas Neo-Brutalist

// 1. Latar Belakang Kertas Grid + Bintang 4-Sudut Berkedip (Sparkles)
export const GridPaperBackground: React.FC<{ children: React.ReactNode }> = ({
  children,
}) => {
  const frame = useCurrentFrame();

  const sparkles = [
    { x: 120, y: 340, color: '#FACC15', size: 28, speed: 0.08 },
    { x: 940, y: 420, color: '#F472B6', size: 24, speed: 0.1 },
    { x: 100, y: 1180, color: '#3B82F6', size: 22, speed: 0.07 },
    { x: 950, y: 1260, color: '#34D399', size: 26, speed: 0.09 },
  ];

  return (
    <AbsoluteFill
      style={{
        backgroundColor: '#F8FAFC',
        backgroundImage: `
          linear-gradient(to right, #E2E8F0 1.5px, transparent 1.5px),
          linear-gradient(to bottom, #E2E8F0 1.5px, transparent 1.5px)
        `,
        backgroundSize: '48px 48px',
        overflow: 'hidden',
        fontFamily: '"Plus Jakarta Sans", "Inter", sans-serif',
        color: INK,
      }}
    >
      {/* Bintang 4-Sudut (Sparkles) */}
      {sparkles.map((sp, i) => {
        const pulse = 0.75 + Math.sin(frame * sp.speed + i * 2) * 0.25;
        const rot = Math.sin(frame * 0.04 + i) * 15;
        return (
          <svg
            key={i}
            width={sp.size}
            height={sp.size}
            viewBox="0 0 24 24"
            style={{
              position: 'absolute',
              left: sp.x,
              top: sp.y,
              transform: `scale(${pulse}) rotate(${rot}deg)`,
            }}
          >
            <path
              d="M12 0 L14.8 9.2 L24 12 L14.8 14.8 L12 24 L9.2 14.8 L0 12 L9.2 9.2 Z"
              fill={sp.color}
            />
          </svg>
        );
      })}
      {children}
    </AbsoluteFill>
  );
};

      }}
    >
      <div style={{ display: 'flex', alignItems: 'center' }}>
        {new Array(totalSteps).fill(0).map((_, idx) => {
          const stepNum = idx + 1;
          const isCompleted = stepNum < currentStep;
          const isActive = stepNum === currentStep;

          const pop = spring({
            frame,
            fps,
            config: { damping: 12, stiffness: 180 },
          });

          const bgColor = isCompleted
            ? '#34D399'
            : isActive
            ? '#FACC15'
            : '#FFFFFF';

          return (
            <React.Fragment key={stepNum}>
              <div
                style={{
                  width: 52,
                  height: 52,
                  borderRadius: '50%',
                  backgroundColor: bgColor,
                  border: `3.5px solid ${INK}`,
                  boxShadow: `3px 3px 0px ${INK}`,
                  display: 'flex',
                  justifyContent: 'center',
                  alignItems: 'center',
                  fontSize: 24,
                  fontWeight: 800,
                  transform: isActive ? `scale(${interpolate(pop, [0, 1], [0.8, 1.08])})` : 'scale(1)',
                }}
              >
                {isCompleted ? '✓' : stepNum}
              </div>
              {stepNum < totalSteps && (
                <div
                  style={{
                    width: 44,
                    height: 6,
                    backgroundColor: isCompleted ? INK : '#CBD5E1',
                  }}
                />
              )}
            </React.Fragment>
          );
        })}
      </div>
    </div>
  );
};

// 4. Transisi Sapuan Miring (Diagonal Color Wipe seperti Detik 00:03 & 00:10 Video Referensi)
export const DiagonalColorWipe: React.FC<{
  triggerAtSecond: number;
  color?: string;
}> = ({ triggerAtSecond, color = '#FACC15' }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const start = Math.round(triggerAtSecond * fps) - 10;

  const progress = interpolate(frame - start, [0, 22], [-130, 130], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  if (frame < start || frame > start + 22) return null;

  return (
    <AbsoluteFill style={{ pointerEvents: 'none', zIndex: 100 }}>
      <div
        style={{
          position: 'absolute',
          top: '-10%',
          bottom: '-10%',
          left: `${progress}%`,
          width: '125%',
          backgroundColor: color,
          borderLeft: `6px solid ${INK}`,
          borderRight: `6px solid ${INK}`,
          transform: 'skewX(-12deg)',
        }}
      />
    </AbsoluteFill>
  );
};
```
